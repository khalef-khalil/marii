%%% FRENCH
\frenchAbstract{Ce mémoire conçoit et évalue \textLR{HAKE-MER} pour la détection multi-étiquettes des émotions sur GoEmotions. À protocole constant (DistilBERT, trois graines ; Step~0 également sur RoBERTa), les ablations tranchent H1 à H4 : M2 (attention par émotion) porte le gain en F1-macro test ($\approx 0{,}505$) ; M1 seul sous-performe le PLM flat ; M3 (NRC, SenticNet) et M4 n'améliorent pas robustement la macro. La configuration retenue est M1+M2. L'analyse par classe montre des progrès surtout sur des émotions de fréquence intermédiaire.}

\frenchAbstractKeywords{Émotions, TALN, GoEmotions, attention hiérarchique, classification multi-étiquettes}

%%% ENGLISH
\englishAbstract{This thesis designs and evaluates \textLR{HAKE-MER} for multi-label emotion detection on GoEmotions. Under a fixed fine-tuning protocol (DistilBERT, three seeds; Step~0 also on RoBERTa), ablations H1 to H4 show that label-specific phrase attention (M2) yields the main test macro F1 gain ($\approx 0.505$), hierarchical encoding alone (M1) underperforms a flat PLM baseline, and lexicon (M3) plus dynamic scaling (M4) do not reliably improve macro F1 over M1+M2. The recommended stack is M1+M2. Per-class analysis indicates gains mainly on mid-frequency emotions, not the rarest labels.}

\englishAbstractKeywords{Emotion detection, NLP, GoEmotions, hierarchical attention, multi-label classification}
